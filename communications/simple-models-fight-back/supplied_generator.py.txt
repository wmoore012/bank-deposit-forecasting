def dotgrid_svg(highlights, color, title, big, sub, x=0, y=0, width=320, dot=12, gap=22, rows=10, cols=10):
    dots = []
    for i in range(rows * cols):
        r = i // cols
        c = i % cols
        fill = color if i < highlights else "#D5D3CD"
        dots.append(f'<circle cx="{x + c*gap}" cy="{y + r*gap}" r="{dot/2}" fill="{fill}"/>')
    return f"""
      <g transform="translate({(width - (cols-1)*gap)/2},0)">
        <text x="{x + (cols-1)*gap/2}" y="{y-34}" text-anchor="middle" fill="{MUTED}" font-size="27" font-weight="900" letter-spacing="0">{title}</text>
        {''.join(dots)}
        <text x="{x + (cols-1)*gap/2}" y="{y + rows*gap + 42}" text-anchor="middle" fill="{color}" font-size="43" font-weight="900">{big}</text>
        <text x="{x + (cols-1)*gap/2}" y="{y + rows*gap + 76}" text-anchor="middle" fill="{MUTED}" font-size="23" font-weight="900">{sub}</text>
      </g>
    """



def zero_hist_svg():
    # Actual central-98% training log-growth distribution extracted from the
    # notebook's target_growth_side_by_side Plotly trace. Units below are
    # percentage points (100 * log_growth); tails remained in the model data.
    counts = [921, 1399, 2389, 4227, 7526, 13715, 23032, 33060, 34073, 28065,
              19857, 13634, 8895, 5846, 3947, 2770, 1938, 1401, 1022, 723,
              580, 441, 367, 307]
    edges = [-10.9688, -9.5678, -8.1668, -6.7658, -5.3647, -3.9637, -2.5627,
             -1.1617, 0.2393, 1.6404, 3.0414, 4.4424, 5.8434, 7.2444, 8.6455,
             10.0465, 11.4475, 12.8485, 14.2495, 15.6506, 17.0516, 18.4526,
             19.8536, 21.2546, 22.6557]
    median = 0.9987
    xmin, xmax = edges[0], edges[-1]
    left, right, top, base = 90, 1040, 40, 280
    maxh = max(counts)
    def sx(v):
        return left + (v - xmin) / (xmax - xmin) * (right-left)
    bars = []
    for i, c in enumerate(counts):
        x1, x2 = sx(edges[i]), sx(edges[i+1])
        h = (base-top-26) * c / maxh
        bars.append(
            f'<rect x="{x1+1:.1f}" y="{base-h:.1f}" width="{max(2, x2-x1-2):.1f}" '
            f'height="{h:.1f}" rx="3" fill="#BEC3C5"/>'
        )
    zero_x = sx(0)
    med_x = sx(median)
    ticks = [-10, 0, 10, 20]
    parts = [
        f'<line x1="{left}" y1="{base}" x2="{right}" y2="{base}" stroke="#D6D4CE" stroke-width="2"/>',
        ''.join(bars),
        f'<line x1="{zero_x:.1f}" y1="{top}" x2="{zero_x:.1f}" y2="{base}" stroke="{CYAN}" stroke-width="6"/>',
        f'<text x="{zero_x:.1f}" y="28" text-anchor="middle" fill="#0A9CB8" font-size="24" font-weight="950">ZERO FORECAST</text>',
        f'<circle cx="{med_x:.1f}" cy="{base}" r="7" fill="{AMBER}" stroke="{CARD_BG}" stroke-width="2"/>',
        f'<text x="{med_x+18:.1f}" y="{base-12}" fill="{AMBER}" font-size="19" font-weight="900">training median ≈ 1.0 pp</text>',
    ]
    for t in ticks:
        x=sx(t)
        parts.append(f'<line x1="{x:.1f}" y1="{base}" x2="{x:.1f}" y2="{base+8}" stroke="#BDBBB5" stroke-width="2"/>')
        parts.append(f'<text x="{x:.1f}" y="{base+34}" text-anchor="middle" fill="{MUTED}" font-size="18" font-weight="800">{t:+g} pp</text>')
    parts.append(f'<text x="{(left+right)/2}" y="352" text-anchor="middle" fill="{MUTED}" font-size="18" font-weight="800">next-quarter log growth · central 98% of training targets shown</text>')
    return f'<svg viewBox="0 0 1130 370" aria-hidden="true">{"".join(parts)}</svg>'




def dotplot_svg(values, labels, colors, xmax, ticks, width=820, height=320):
    # Honest zero-baseline scale sized for a side-by-side panel.
    # Extra left padding prevents long labels from clipping in the exported PDF.
    left, right, top, gap_y = 290, 790, 52, 58
    def scale(v):
        return left + v / xmax * (right-left)
    base_y = top + gap_y*(len(values)-1) + 38
    parts = [f'<line x1="{left}" y1="{base_y}" x2="{right}" y2="{base_y}" stroke="{GRID}" stroke-width="2"/>']
    for t in ticks:
        x=scale(t)
        parts.append(f'<line x1="{x}" y1="30" x2="{x}" y2="{base_y}" stroke="#F0EEE8" stroke-width="1.5"/>')
        parts.append(f'<text x="{x}" y="{height-18}" text-anchor="middle" fill="{MUTED}" font-size="20" font-weight="800">{t:g}</text>')
    for i,(v,lab,col) in enumerate(zip(values,labels,colors)):
        y=top+i*gap_y
        x=scale(v)
        is_first=i==0
        parts.append(f'<text x="{left-22}" y="{y+8}" text-anchor="end" fill="{INK if is_first else MUTED}" font-size="26" font-weight="{950 if is_first else 850}">{lab}</text>')
        parts.append(f'<line x1="{left}" y1="{y}" x2="{x}" y2="{y}" stroke="#D8D6D0" stroke-width="7" stroke-linecap="round"/>')
        parts.append(f'<circle cx="{x}" cy="{y}" r="{14 if is_first else 11}" fill="{col}" stroke="{CARD_BG}" stroke-width="2"/>')
        tx=x+20; anchor='start'
        if tx > right-52:
            tx=x-18; anchor='end'
        parts.append(f'<text x="{tx}" y="{y+9}" text-anchor="{anchor}" fill="{INK if is_first else MUTED}" font-size="25" font-weight="{950 if is_first else 850}">{v:.3f}</text>')
    return f'<svg viewBox="0 0 {width} {height}" aria-hidden="true">{"".join(parts)}</svg>'

HTML = f"""<!doctype html><html><head><meta charset="utf-8"/><meta name="viewport" content="width=1792, initial-scale=1"/><title>Simple Models Fight Back</title><style>{CSS}</style></head><body>{BODY}</body></html>"""


async def render_pages():
    HTML_FILE.write_text(HTML, encoding="utf-8")
    if RENDER_DIR.exists():
        shutil.rmtree(RENDER_DIR)
    RENDER_DIR.mkdir(exist_ok=True)

    async with async_playwright() as p:
        chromium_path = shutil.which("chromium") or shutil.which("google-chrome") or shutil.which("chromium-browser")
        launch_kwargs = {"args": ["--no-sandbox"]}
        if chromium_path:
            launch_kwargs["executable_path"] = chromium_path
        browser = await p.chromium.launch(**launch_kwargs)
        page = await browser.new_page(viewport={"width": 1792, "height": 1024}, device_scale_factor=1.25)
        await page.set_content(HTML_FILE.read_text(encoding="utf-8"), wait_until="networkidle")
        cards = await page.locator(".page").all()
        for i, card in enumerate(cards, start=1):
            await card.screenshot(path=str(RENDER_DIR / f"page-{i}.png"))
        await browser.close()


def build_pptx():
    prs = Presentation()
    prs.slide_width = Inches(14)
    prs.slide_height = Inches(8)
    blank_layout = prs.slide_layouts[6]
    for img in sorted(RENDER_DIR.glob("page-*.png")):
        slide = prs.slides.add_slide(blank_layout)
        slide.shapes.add_picture(str(img), 0, 0, width=prs.slide_width, height=prs.slide_height)
    prs.save(PPTX_FILE)


def build_pdf():
    pages = []
    for img in sorted(RENDER_DIR.glob("page-*.png")):
        pages.append(Image.open(img).convert("RGB"))
    if not pages:
        raise RuntimeError("No rendered PNG pages found.")
    pages[0].save(PDF_FILE, save_all=True, append_images=pages[1:], resolution=160.0)


def main():
    asyncio.run(render_pages())
    build_pptx()
    build_pdf()
    print(f"Wrote {HTML_FILE}")
    print(f"Wrote {PPTX_FILE}")
    print(f"Wrote {PDF_FILE}")


if __name__ == "__main__":
    main()
